Environment template for Supabase and preview protection.

.env.example 
# Supabase (Project Settings -> API)
NEXT_PUBLIC_SUPABASE_URL=https://YOUR-PROJECT.supabase.co
NEXT_PUBLIC_SUPABASE_ANON_KEY=YOUR-ANON-KEY

# Server-only (API route / Edge Function deploys). Never exposed to the client.
SUPABASE_SERVICE_ROLE_KEY=YOUR-SERVICE-ROLE-KEY

# Vercel preview protection password until community engagement is complete
PREVIEW_PASSWORD=choose-something
